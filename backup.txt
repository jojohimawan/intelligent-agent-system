func decodeOBD2Frame(frame can.Frame, msg *bit11can.OBD2) error {
	if err := msg.UnmarshalFrame(frame); err != nil {
		fmt.Printf("   Raw data: % X\n", frame.Data[:frame.Length])
		return fmt.Errorf("⚠️  Failed to unmarshal frame: %v", err)
	}

	msg.Descriptor()

	service := msg.Service()
	response := msg.Response()
	fmt.Printf("📋 Service raw: %d (0x%02X)\n", uint8(service), uint8(service))
	fmt.Printf("📋 S01PID raw: %d (0x%02X)\n", uint8(msg.S01PID()), uint8(msg.S01PID()))
	fmt.Printf("📋 Service: 0x%02X, Response: 0x%02X\n", service, response)

	if uint8(service) != 1 {
		return fmt.Errorf("unsupported service: 0x%02X", service)
	}

	A := frame.Data[3]
	B := frame.Data[4]
	rpm := float64(uint16(A)<<8|uint16(B)) / 4.0
	fmt.Printf("🔧 Engine RPM (manual decode): %.0f\n", rpm)

	return decodeMode01PID(msg)
}

func decodeMode01PID(msg *bit11can.OBD2) error {
	pid := msg.S01PID()

	fmt.Print("\nPID: ")
	fmt.Println(pid.String())

	fmt.Print("\nDescription: ")
	fmt.Println(msg.Descriptor().Description)

	fmt.Print("\nID: ")
	fmt.Println(msg.Descriptor().ID)

	fmt.Print("\nExtended: ")
	fmt.Println(msg.Descriptor().IsExtended)

	fmt.Print("\nLength: ")
	fmt.Println(msg.Descriptor().Length)

	fmt.Print("\nName: ")
	fmt.Println(msg.Descriptor().Name)

	fmt.Print("\nSend type: ")
	fmt.Println(msg.Descriptor().SendType)

	fmt.Print("\nSignals: ")
	fmt.Println(msg.Descriptor().Signals)

	fmt.Print("\nMultiplexer signal: ")
	fmt.Println(msg.Descriptor().MultiplexerSignal())

	fmt.Print("\nFrame data: ")
	fmt.Println(msg.Frame().Data)

	if pid != bit11can.OBD2_S01PID_S01PID0CEngineRPM {
		return fmt.Errorf("unhandled Mode 01 PID: 0x%02X", pid)
	}

	rpm := msg.S01PID0C_EngineRPM()
	fmt.Printf("RPM: %.2f\n", rpm)
	actualRpm := float64(rpm) / 4.0
	fmt.Printf("🔧 Engine RPM: %.0f\n", actualRpm)

	return nil
}

func defineSignals() {
	md := bit11can.Messages().OBD2
	fmt.Println("=== OBD2 Signal Definitions ===")

	fmt.Printf("Length Signal:\n")
	fmt.Printf("  Start: %d, Length: %d, BigEndian: %v\n\n",
		md.Length.Start, md.Length.Length, md.Length.IsBigEndian)

	fmt.Printf("Service Signal:\n")
	fmt.Printf("  Start: %d, Length: %d, BigEndian: %v\n\n",
		md.Service.Start, md.Service.Length, md.Service.IsBigEndian)

	fmt.Printf("S01PID Signal:\n")
	fmt.Printf("  Start: %d, Length: %d, BigEndian: %v\n\n",
		md.S01PID.Start, md.S01PID.Length, md.S01PID.IsBigEndian)

	fmt.Printf("S01PID0C_EngineRPM Signal:\n")
	fmt.Printf("  Start: %d, Length: %d, BigEndian: %v\n",
		md.S01PID0C_EngineRPM.Start, md.S01PID0C_EngineRPM.Length,
		md.S01PID0C_EngineRPM.IsBigEndian)
	fmt.Printf("  Scale: %f, Offset: %f\n",
		md.S01PID0C_EngineRPM.Scale, md.S01PID0C_EngineRPM.Offset)
}
